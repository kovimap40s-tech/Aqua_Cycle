/* 
 * KODE PROGRAM ESP32 - SIMULASI SIKLUS AIR (FIREBASE REALTIME DATABASE)
 * Terhubung ke Web App HTML5 Simulasi Siklus Air
 * Sensor: DHT22, Sensor Hujan Digital (DO), Sensor IR (Digital)
 * Actuator: Relay LED (Matahari), Relay Mist Maker, MOSFET Pompa Air, LCD I2C
 */

#include <WiFi.h>
#include <FirebaseESP32.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>

// ===================== KONEKSI & FIREBASE CONFIG =====================
#define WIFI_SSID "Axez"
#define WIFI_PASSWORD "qazxswedc"

//database dirahasiakan!!

// ===================== PIN ESP32 =====================
#define PIN_DHT          16   // DHT22 Data
#define PIN_RELAY_LED    23   // Relay LED Matahari
#define PIN_RELAY_MIST   19   // Relay Mist Maker
#define PIN_POMPA        18   // MOSFET Pompa Air
#define PIN_RAIN_DIGITAL 15   // Rain Sensor Pin Digital (DO)
#define PIN_IR           4    // Sensor IR Digital
#define PIN_SDA          21   // LCD SDA
#define PIN_SCL          22   // LCD SCL

const bool RELAY_AKTIF_LOW = true;
const float SUHU_TARGET_MIN = 30.0;
const float SUHU_TARGET_MAX = 32.0;

#define DHTTYPE DHT22
DHT dht(PIN_DHT, DHTTYPE);
LiquidCrystal_I2C lcd(0x27, 16, 2);

FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

// Variabel Sensor Realtime
float suhu = 0.0;
float hum = 0.0;
bool rainDigital = false; // HIGH/LOW
bool irTerdeteksi = false;

// Status Output Hardware
bool ledState = false;
bool mistState = false;
bool pompaState = false;

// Multi-mode System Variables
int activeMode = 1; // 1 = Read Sensor, 2 = Manual Mode, 3 = Auto Sim

unsigned long tSensors = 0, tFirebase = 0, tLCD = 0;

void relayTulis(uint8_t pin, bool nyala) {
  bool level = RELAY_AKTIF_LOW ? !nyala : nyala;
  digitalWrite(pin, level ? HIGH : LOW);
}

void setup() {
  Serial.begin(115200);

  pinMode(PIN_RELAY_LED, OUTPUT);
  pinMode(PIN_RELAY_MIST, OUTPUT);
  pinMode(PIN_POMPA, OUTPUT);
  pinMode(PIN_RAIN_DIGITAL, INPUT);
  pinMode(PIN_IR, INPUT);

  relayTulis(PIN_RELAY_LED, false);
  relayTulis(PIN_RELAY_MIST, false);
  digitalWrite(PIN_POMPA, LOW);

  Wire.begin(PIN_SDA, PIN_SCL);
  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0); lcd.print("Siklus Air ESP32");
  lcd.setCursor(0, 1); lcd.print("Connecting WiFi..");

  dht.begin();

  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected!");

  config.host = FIREBASE_HOST;
  config.signer.tokens.legacy_token = FIREBASE_AUTH;
  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);

  lcd.clear();
  lcd.setCursor(0, 0); lcd.print("Firebase Ready!");
  delay(1500);
}

void loop() {
  unsigned long now = millis();

  // 1. Baca Sensor lokal tiap 1 detik
  if (now - tSensors >= 1000) {
    tSensors = now;
    float t = dht.readTemperature();
    float h = dht.readHumidity();
    if (!isnan(t)) suhu = t;
    if (!isnan(h)) hum = h;

    // Digital Rain Sensor: LOW = Hujan/Basah pada modul konvensional (Pin DO)
    rainDigital = (digitalRead(PIN_RAIN_DIGITAL) == LOW);
    irTerdeteksi = (digitalRead(PIN_IR) == HIGH);
  }

  // 2. Baca Mode Aktif & Proses Logika Firebase (Tiap 400ms)
  if (now - tFirebase >= 400) {
    tFirebase = now;

    if (Firebase.getInt(fbdo, "/system/active_mode")) {
      if (fbdo.dataType() == "int") {
        activeMode = fbdo.intData();
      }
    }

    if (activeMode == 1) {
      // ===== MODE 1: READ SENSOR MODE =====
      // Logika otomatis berdasarkan suhu (30-32 C) dan kelembaban (>80%)
      if (suhu >= SUHU_TARGET_MIN && suhu <= SUHU_TARGET_MAX) {
        ledState = true;
        mistState = true;
      } else if (suhu > SUHU_TARGET_MAX) {
        ledState = false;
        mistState = false;
      } else if (suhu < SUHU_TARGET_MIN) {
        ledState = true;
        mistState = false;
      }

      pompaState = (hum >= 80.0);

      // Tulis data sensor lokal ke Firebase (/mode1_sensor)
      Firebase.setFloat(fbdo, "/mode1_sensor/suhu", suhu);
      Firebase.setFloat(fbdo, "/mode1_sensor/kelembaban", hum);
      Firebase.setBool(fbdo, "/mode1_sensor/led", ledState);
      Firebase.setBool(fbdo, "/mode1_sensor/mist", mistState);
      Firebase.setBool(fbdo, "/mode1_sensor/pompa", pompaState);
      Firebase.setBool(fbdo, "/mode1_sensor/rain_digital", rainDigital);
      Firebase.setBool(fbdo, "/mode1_sensor/ir", irTerdeteksi);

    } else if (activeMode == 2) {
      // ===== MODE 2: MANUAL MODE =====
      // Baca perintah langsung dari website (/mode2_manual)
      if (Firebase.getBool(fbdo, "/mode2_manual/led")) ledState = fbdo.boolData();
      if (Firebase.getBool(fbdo, "/mode2_manual/mist")) mistState = fbdo.boolData();
      if (Firebase.getBool(fbdo, "/mode2_manual/pompa")) pompaState = fbdo.boolData();

    } else if (activeMode == 3) {
      // ===== MODE 3: AUTO SIMULATION MODE =====
      // Baca status yang dikirim animasi web (/mode3_auto)
      if (Firebase.getBool(fbdo, "/mode3_auto/led")) ledState = fbdo.boolData();
      if (Firebase.getBool(fbdo, "/mode3_auto/mist")) mistState = fbdo.boolData();
      if (Firebase.getBool(fbdo, "/mode3_auto/pompa")) pompaState = fbdo.boolData();
    }

    // Eksekusi Output Relay & Mosfet Hardware
    relayTulis(PIN_RELAY_LED, ledState);
    relayTulis(PIN_RELAY_MIST, mistState);
    digitalWrite(PIN_POMPA, pompaState ? HIGH : LOW);
  }

  // 3. Update Tampilan LCD 16x2
  if (now - tLCD >= 500) {
    tLCD = now;
    lcd.setCursor(0, 0);
    lcd.print("M:"); lcd.print(activeMode);
    lcd.print(" T:"); lcd.print(suhu, 1);
    lcd.print("C H:"); lcd.print((int)hum); lcd.print("%  ");

    lcd.setCursor(0, 1);
    lcd.print("L:"); lcd.print(ledState ? "1" : "0");
    lcd.print(" M:"); lcd.print(mistState ? "1" : "0");
    lcd.print(" P:"); lcd.print(pompaState ? "1" : "0");
    lcd.print(" IR:"); lcd.print(irTerdeteksi ? "Y" : "N");
    lcd.print("  ");
  }
}
